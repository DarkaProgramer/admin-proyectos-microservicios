3. Dockerfiles Multi-Stage (Compilación Java 21 y Runtime Ligero)

Ambos proyectos utilizan Multi-Stage Build para garantizar un tamaño mínimo de imagen y máxima seguridad en producción:

  • Etapa 1 (Build): maven:3.9-eclipse-temurin-21 empaqueta el artefacto .jar con Java 21.
  • Etapa 2 (Runtime): eclipse-temurin:21-jre ejecuta el .jar limitando memoria con -Xmx256m.
# Etapa 1: Compilación
FROM maven:3.9-eclipse-temurin-21 AS build
WORKDIR /app
COPY pom.xml .
COPY src ./src
RUN mvn -B -q clean package -DskipTests

# Etapa 2: Runtime ligero
FROM eclipse-temurin:21-jre
WORKDIR /app
COPY --from=build /app/target/*.jar app.jar
ENV JAVA_OPTS="-Xmx256m"
ENTRYPOINT ["sh", "-c", "java $JAVA_OPTS -jar app.jar"]

4. Orquestador docker-compose.yml

services:
  db:
    image: mariadb:11.3
    container_name: gestor-db
    environment:
      MARIADB_ROOT_PASSWORD: root123
      MARIADB_DATABASE: dbgestor
    volumes:
      - ./docker/db/init:/docker-entrypoint-initdb.d:ro
      - db-data:/var/lib/mysql
    healthcheck:
      test: ["CMD", "healthcheck.sh", "--connect", "--innodb_initialized"]
      interval: 10s
      timeout: 5s
      retries: 12
      start_period: 20s

  backend:
    build: ./Ejercicio5MyAPIGestor
    image: gestor-backend:1.0
    container_name: gestor-backend
    environment:
      SPRING_DATASOURCE_URL: jdbc:mariadb://db:3306/dbgestor
      SPRING_DATASOURCE_USERNAME: root
      SPRING_DATASOURCE_PASSWORD: root123
      JAVA_OPTS: "-Xmx256m"
    depends_on:
      db:
        condition: service_healthy
    ports:
      - "8081:8081"

  frontend:
    build: ./Ejercicio5Cliente
    image: gestor-frontend:1.0
    container_name: gestor-frontend
    environment:
      BACKEND_URL: http://backend:8081
      JAVA_OPTS: "-Xmx256m"
    depends_on:
      - backend
    ports:
      - "9080:9080"

volumes:
  db-data:
